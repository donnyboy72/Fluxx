19/// The standard handles of a process.
20///
21/// See [the Windows documentation on console
22/// handles](https://docs.microsoft.com/en-us/windows/console/console-handles) for more info.
23#[derive(Debug, Clone, Copy)]
24pub enum HandleType {
25    /// The process' standard output.
26    OutputHandle,
27    /// The process' standard input.
28    InputHandle,
29    /// The process' active console screen buffer, `CONOUT$`.
30    CurrentOutputHandle,
31    /// The process' console input buffer, `CONIN$`.
32    CurrentInputHandle,
33}
34
35/// Inner structure for closing a handle on Drop.
36///
37/// The second parameter indicates if the HANDLE is exclusively owned or not.
38/// A non-exclusive handle can be created using for example
39/// `Handle::input_handle` or `Handle::output_handle`, which corresponds to
40/// stdin and stdout respectively.
41#[derive(Debug)]
42struct Inner {
43    handle: HANDLE,
44    is_exclusive: bool,
45}
46
47impl Inner {
48    fn new_exclusive(handle: HANDLE) -> Self {
49        Inner {
50            handle,
51            is_exclusive: true,
52        }
53    }
54
55    fn new_shared(handle: HANDLE) -> Self {
56        Inner {
57            handle,
58            is_exclusive: false,
59        }
60    }
61}
62
63impl Drop for Inner {
64    fn drop(&mut self) {
65        if self.is_exclusive {
66            assert!(
67                unsafe { CloseHandle(self.handle) != 0 },
68                "failed to close handle"
69            )
70        }
71    }
72}
73
74unsafe impl Send for Inner {}
75
76unsafe impl Sync for Inner {}

78/// This abstracts away some WinAPI calls to set and get some console handles.
79///
80/// It wraps WinAPI's [`HANDLE`] type.
81#[derive(Debug, Clone)]
82pub struct Handle {
83    handle: Arc<Inner>,
84}
85
86impl Handle {
87    /// Create a new handle of a certaint type.
88    pub fn new(handle: HandleType) -> Result<Handle> {
89        match handle {
90            HandleType::OutputHandle => Handle::output_handle(),
91            HandleType::InputHandle => Handle::input_handle(),
92            HandleType::CurrentOutputHandle => Handle::current_out_handle(),
93            HandleType::CurrentInputHandle => Handle::current_in_handle(),
94        }
95    }
96
97    /// Construct a handle from a raw handle.
98    ///
99    /// # Safety
100    ///
101    /// This is unsafe since there is not guarantee that the underlying HANDLE is thread-safe to implement `Send` and `Sync`.
102    /// Most HANDLE's however, are thread safe.
103    pub unsafe fn from_raw(handle: HANDLE) -> Self {
104        Self {
105            handle: Arc::new(Inner::new_exclusive(handle)),
106        }
107    }
108
109    /// Get the handle of the active screen buffer.
110    /// When using multiple screen buffers this will always point to the to the current screen output buffer.
111    ///
112    /// This function uses `CONOUT$` to create a file handle to the current output buffer.
113    ///
114    /// This wraps
115    /// [`CreateFileW`](https://docs.microsoft.com/en-us/windows/win32/api/fileapi/nf-fileapi-createfilew).
116    pub fn current_out_handle() -> Result<Handle> {
117        let utf16: Vec<u16> = "CONOUT$\0".encode_utf16().collect();
118        let utf16_ptr: *const u16 = utf16.as_ptr();
119
120        let handle = handle_result(unsafe {
121            CreateFileW(
122                utf16_ptr,
123                GENERIC_READ | GENERIC_WRITE,
124                FILE_SHARE_READ | FILE_SHARE_WRITE,
125                null_mut(),
126                OPEN_EXISTING,
127                0,
128                null_mut(),
129            )
130        })?;
131
132        Ok(Handle {
133            handle: Arc::new(Inner::new_exclusive(handle)),
134        })
135    }
136
137    /// Get the handle of the console input buffer.
138    ///
139    /// This function uses `CONIN$` to create a file handle to the current input buffer.
140    ///
141    /// This wraps
142    /// [`CreateFileW`](https://docs.microsoft.com/en-us/windows/win32/api/fileapi/nf-fileapi-createfilew).
143    pub fn current_in_handle() -> Result<Handle> {
144        let utf16: Vec<u16> = "CONIN$\0".encode_utf16().collect();
145        let utf16_ptr: *const u16 = utf16.as_ptr();
146
147        let handle = handle_result(unsafe {
148            CreateFileW(
149                utf16_ptr,
150                GENERIC_READ | GENERIC_WRITE,
151                FILE_SHARE_READ | FILE_SHARE_WRITE,
152                null_mut(),
153                OPEN_EXISTING,
154                0,
155                null_mut(),
156            )
157        })?;
158
159        Ok(Handle {
160            handle: Arc::new(Inner::new_exclusive(handle)),
161        })
162    }
163
164    /// Get the handle of the standard output.
165    ///
166    /// On success this function returns the `HANDLE` to `STD_OUTPUT_HANDLE`.
167    ///
168    /// This wraps [`GetStdHandle`](https://docs.microsoft.com/en-us/windows/console/getstdhandle)
169    /// called with `STD_OUTPUT_HANDLE`.
170    pub fn output_handle() -> Result<Handle> {
171        Self::std_handle(STD_OUTPUT_HANDLE)
172    }
173
174    /// Get the handle of the input screen buffer.
175    ///
176    /// On success this function returns the `HANDLE` to `STD_INPUT_HANDLE`.
177    ///
178    /// This wraps [`GetStdHandle`](https://docs.microsoft.com/en-us/windows/console/getstdhandle)
179    /// called with `STD_INPUT_HANDLE`.
180    pub fn input_handle() -> Result<Handle> {
181        Self::std_handle(STD_INPUT_HANDLE)
182    }
183
184    fn std_handle(which_std: DWORD) -> Result<Handle> {
185        let handle = handle_result(unsafe { GetStdHandle(which_std) })?;
186
187        Ok(Handle {
188            handle: Arc::new(Inner::new_shared(handle)),
189        })
190    }
191
192    /// Checks if the console handle is an invalid handle value.
193    ///
194    /// This is done by checking if the passed `HANDLE` is equal to `INVALID_HANDLE_VALUE`.
195    pub fn is_valid_handle(handle: &HANDLE) -> bool {
196        *handle != INVALID_HANDLE_VALUE
197    }
198}
199
200impl Deref for Handle {
201    type Target = HANDLE;
202
203    fn deref(&self) -> &HANDLE {
204        &self.handle.handle
205    }
206}
