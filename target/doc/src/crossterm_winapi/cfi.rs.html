8/// Information about the font.

12#[derive(Clone)]
13pub struct FontInfo(pub CONSOLE_FONT_INFO);
14

18        f.debug_struct("FontInfo")

20            .field("nFont", &self.index())

22    }

25impl FontInfo {
26    /// Create a new font info without all zeroed properties.
27    pub fn new() -> FontInfo {
28        FontInfo(unsafe { zeroed() })
29    }
30
31    /// Get the size of the font.
32    ///
33    /// Will take `dwFontSize` from the current font info and convert it into a [`Size`].
34    pub fn size(&self) -> Size {
35        Size::from(self.0.dwFontSize)
36    }
37
38    /// Get the index of the font in the system's console font table.
39    pub fn index(&self) -> u32 {
40        self.0.nFont
41    }
